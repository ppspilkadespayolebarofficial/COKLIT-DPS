function rapikanDanWarnaiCoklit() { var ss = SpreadsheetApp.getActiveSpreadsheet(); var sheet = ss.getSheetByName("Hasil Coklit"); if (!sheet) { throw new Error('Sheet "Hasil Coklit" tidak ditemukan.'); } var lastRow = sheet.getLastRow(); var lastColumn = Math.max(sheet.getLastColumn(), 13); if (lastRow <= 1) return; var data = sheet.getRange(2, 1, lastRow - 1, lastColumn).getValues(); if (data.length > 0) { var newRange = sheet.getRange(2, 1, data.length, lastColumn); var backgrounds = []; var currentColorIndex = 0; var lastNkk = ""; var colors = ["#ffffff", "#f1f5f9"]; // Putih & Abu-abu sangat muda for (var r = 0; r < data.length; r++) { var currentNkk = String(data[r][2] || "").trim(); // Kolom C = No KK if (currentNkk === "") { var rowColors = []; for (var c = 0; c < lastColumn; c++) { rowColors.push(colors[currentColorIndex]); } backgrounds.push(rowColors); continue; } if (r === 0) { lastNkk = currentNkk; } else if (currentNkk !== lastNkk) { currentColorIndex = 1 - currentColorIndex; lastNkk = currentNkk; } var rowColors = []; for (var c = 0; c < lastColumn; c++) { rowColors.push(colors[currentColorIndex]); } backgrounds.push(rowColors); } newRange.setBackgrounds(backgrounds); } }
